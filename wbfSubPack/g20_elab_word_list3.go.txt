func sortWordListByFreq_and_row_priority() {
	
	fmt.Println(" sortWordListByFreq_and_row_priority()")
	
	sort.Slice(wordSliceAlpha, func(i, j int) bool {
		
		if wordSliceAlpha[i].wTotRow !=  wordSliceAlpha[j].wTotRow {
		   return wordSliceAlpha[i].wTotRow > wordSliceAlpha[j].wTotRow         // totRow    descending order (how many rows contain the word)  	  		   
		}	
		if wordSliceAlpha[i].wTotExtrRow !=  wordSliceAlpha[j].wTotExtrRow {
		   return wordSliceAlpha[i].wTotExtrRow > wordSliceAlpha[j].wTotExtrRow  // wTotExtrRow  descending order ( number of extracted rows  if id_sel_2_extrRow option = id="extrRow", else =0) 	
		}
		
		if wordSliceAlpha[i].wWordSeq !=  wordSliceAlpha[j].wWordSeq {
		   return wordSliceAlpha[i].wWordSeq < wordSliceAlpha[j].wWordSeq            // word      ascending order	  		   
		}	
		if wordSliceAlpha[i].wWord2 !=  wordSliceAlpha[j].wWord2 {
		   return wordSliceAlpha[i].wWord2 < wordSliceAlpha[j].wWord2                // word      ascending order	  		   
		}				
		if wordSliceAlpha[i].wTotMinRow !=  wordSliceAlpha[j].wTotMinRow {
		   return wordSliceAlpha[i].wTotMinRow < wordSliceAlpha[j].wTotMinRow  // totMinRow ascending order	(how many words in the row are not yet learned) 
		}
		if wordSliceAlpha[i].wSwSelRowR !=  wordSliceAlpha[j].wSwSelRowR {
		   return wordSliceAlpha[i].wSwSelRowR < wordSliceAlpha[j].wSwSelRowR    // first the extracted row  
		}
		return wordSliceAlpha[i].wIxRow < wordSliceAlpha[j].wIxRow             // ixRow     ascending order	( first the rows which were first in the text)  		   
			
	})	
	
	/**
		fmt.Println( "\nANTONIO5 wordSliceAlpha -------------------------")
	for g:=0; g < len( wordSliceAlpha ); g++ {
		fmt.Println( "ANTONIO5 alpha ", wordSliceAlpha[g] )
	}	
	**/
	
} // end of sortWordListByFreq_and_row_priority
